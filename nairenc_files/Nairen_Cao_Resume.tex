\section{Selected Research Experience}
\project{Cost-Aware LLM Evaluation}{2026}
\begin{itemize}
\item Designed and implemented experiments for cost-aware LLM evaluation using Bayesian bandits; supervised students in running the experiments and comparing adaptive evaluation policies.
\item Evaluated a Gittins-index policy combining posterior uncertainty, adaptive sampling, and a stopping rule; offline index precomputation supports efficient online decisions under unequal evaluation costs.
\item On GSM8K, PIQA, and MMLU response matrices, the reported experiments often achieved near-zero simple regret at stopping with \textbf{about 1--10\% of the exhaustive evaluation budget}; comparisons included Bayesian optimization and frequentist bandit baselines.
\end{itemize}
{\small\textbf{Recognition:} ICML 2026 DEMO Workshop Spotlight; publication details on page 2.}\par
\vspace{4pt}
\project{Scalable Correlation Clustering}{2024--2026}
\begin{itemize}
\item Main contributor to theory across \textbf{six papers} on correlation clustering: cluster linear programs, min-max and simultaneous-norm objectives, parallel algorithms, and dynamic updates.
\item Joint results include a (3 + ε) min-max approximation in nearly linear time in positive edges and a (1.485 + ε) total-disagreement approximation with nearly linear dependence on vertices for fixed ε; developed approaches for scalable computation.
\end{itemize}
\vspace{4pt}
\project{Parallel and Distributed Graph Algorithms}{2019--2024}
\begin{itemize}
\item Co-authored eight conference and journal publications on algorithms for exact and approximate shortest paths, directed hopsets, reachability, and I/O-efficient topological sorting, including graphs with negative edge weights.
\item Joint exact shortest-path work achieved almost linear work and approximately square-root parallel depth for non-negative integer weights; negative-weight shortest-path research received a \textbf{2022 Outstanding Paper Award}.
\end{itemize}
